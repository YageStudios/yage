const fs = require("node:fs");
const path = require("node:path");
const { createCoverageMap, createFileCoverage } = require("/home/robert/.npm/_npx/b8d86e6551a4f492/node_modules/istanbul-lib-coverage");
const { createSourceMapStore } = require("/home/robert/.npm/_npx/b8d86e6551a4f492/node_modules/istanbul-lib-source-maps");

(async () => {
  const directory = "/tmp/far-bake-prototype/coverage-final";
  const merged = createCoverageMap({});
  for (const file of fs.readdirSync(directory)) {
    if (file.endsWith(".test.ts.json")) merged.merge(JSON.parse(fs.readFileSync(path.join(directory, file), "utf8")));
  }
  const remapped = await createSourceMapStore().transformCoverage(merged);
  const file = remapped.files().find(file => file.endsWith("/engine/streaming/far.ts"));
  const coverage = remapped.fileCoverageFor(file);
  const lines = fs.readFileSync(file, "utf8").split("\n");
  const from = lines.findIndex(line => line.includes("// One submission keeps adjacent pieces")) + 1;
  const to = lines.findIndex(line => line.includes("// Each instance's occluder once")) + 1;
  if (from <= 0 || to <= from) throw new Error("changed rendering block not found");
  const data = coverage.toJSON();
  const selected = { path: file, statementMap: {}, fnMap: {}, branchMap: {}, s: {}, f: {}, b: {} };
  for (const [counts, locations] of [["s", "statementMap"], ["f", "fnMap"], ["b", "branchMap"]]) {
    for (const [id, location] of Object.entries(data[locations])) {
      const line = (location.loc ?? location).start.line;
      if (line < from || line >= to) continue;
      selected[locations][id] = location;
      selected[counts][id] = data[counts][id];
    }
  }
  const summary = {
    file,
    wholeModule: coverage.toSummary().toJSON(),
    changedRenderingBlock: { from, toExclusive: to, summary: createFileCoverage(selected).toSummary().toJSON() },
  };
  fs.writeFileSync(path.join(directory, "merged-remapped.json"), JSON.stringify(remapped.toJSON()));
  fs.writeFileSync(path.join(directory, "summary.json"), JSON.stringify(summary, null, 2));
  console.log(JSON.stringify(summary, null, 2));
})().catch(error => { console.error(error); process.exitCode = 1; });
