      depth--;

      if (outer) { const ms = performance.now() - at; m.render.push(ms); m.renderRecords.push({ at, ms, screen }); m.renderTotal += ms;
        if (screen) m.screenRecords.push({ at, ms, insideModel: m.activeRun });
      }
    }
  };
  if (!light) for (const method of ["texImage2D", "texSubImage2D"]) {
    const real = gl[method];

  }
